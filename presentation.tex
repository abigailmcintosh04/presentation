% ========== TOP INTRODUCTION ==========
\begin{frame}{Top Quark}
  \begin{itemize}
    \item Heaviest known elementary particle -- $m_t = \qty{172.76 \pm 0.3}{\giga\electronvolt}$.
    \item Yukawa coupling $y_t \sim 1$, largest of all fermions.
    \item Short lifetime ($\tau \sim \qty{5e-25}{\second}$), decays before hadronisation.
    \item Main production mechanisms:
    \begin{itemize}
      \item Pair production via strong interaction.
      \item Single top production via electroweak interaction (three channels).
      \item Associated production with other particles.
    \end{itemize}
    \item Plays large role in electroweak symmetry breaking and sensitive to BSM physics.
    \item LHC is a top factory.
  \end{itemize}
\end{frame}

% ========== ASSOCIATED TOP PRODUCTION ==========
\begin{frame}{Associated Top Production}
  \begin{itemize}
    \item Top quarks can also be produced in association with other particles, such as:
    \begin{itemize}

  \end{itemize}

\begin{frame}{$t\bar{t}W$ Cross Section}
  \begin{itemize}
    \item ATLAS: Inclusive cross section $\sigma_{t\bar{t}W} = \qty{880(80)}{\femto\barn}$, in agreement with SM prediction within 1.4$\sigma$.
    \begin{itemize}
      \item Run 2 of LHC, $\sqrt{s} = \qty{13}{\tera\electronvolt}$, $L_{int} = \qty{140}{\per\femto\barn}$.
      \item Relative charge asymmetry also agrees with SM prediction.
    \end{itemize}
    \item CMS: Inclusive cross section $\sigma_{t\bar{t}W} = 868 \pm 40 \text{(stat)} \pm 51 \text{(syst)} \unit{\femto\barn}$, within 2$\sigma$ of SM prediction. 
    \begin{itemize}
      \item Run 2 of LHC, $\sqrt{s} = \qty{13}{\tera\electronvolt}$, $L_{int} = \qty{138}{\per\femto\barn}$.
      \item Ratio of $\sigma_{t\bar{t}W^+}$ to $\sigma_{t\bar{t}W^-}$ measured to be $1.61 \pm 0.15\,\text{(stat)}^{+0.07}_{-0.05}\,\text{(syst)}$, consistent with SM within 2$\sigma$.
    \end{itemize}
  \end{itemize}
    {\tiny \color{gray} ATLAS Collaboration, \textbf{\href{https://doi.org/10.1007/JHEP07(2023)219}{JHEP 07 (2023) 219}} [\href{https://arxiv.org/abs/2208.06485v2}{\text{arXiv:2208.06485}}]}
\end{frame}

  \begin{itemize}
    \item $t\bar{t}H$ production observed in many different Higgs decay channels, including $H \rightarrow b\bar{b}$, $H \rightarrow \gamma\gamma$, $H \rightarrow ZZ^* \rightarrow 4\ell$, and $H \rightarrow \tau^+\tau^-$.
    \begin{itemize}
      \item All results compatible with SM predictions.
    \end{itemize}
    \item Analysis of six multilepton channels peformed by ATLAS in 2026, with inclusive cross section measured at $\sigma_{t\bar{t}H} = (321\,^{+102}_{-99})\,\unit{\femto\barn}$, within 2$\sigma$ of SM prediction.
    \begin{itemize}
      \item ATLAS, LHC Run 2, $\sqrt{s} = \qty{13}{\tera\electronvolt}$, $L_\text{int} = \qty{140}{\per\femto\barn}$.
    \end{itemize}
