% ========== TOP INTRODUCTION ==========
\begin{frame}{Top Quark}
  \begin{itemize}
    \item Heaviest known elementary particle ($m_t = 172.76 \pm 0.3$ GeV).
    \item Yukawa coupling $y_t \sim 1$, largest of all fermions.
    \item Short lifetime ($\tau \sim 5 \times 10^{-25}$ s), decays before hadronisation.
    \item Main production mechanisms:
    \begin{itemize}
      \item Pair production via strong interaction.
      \item Single top production via electroweak interaction (three channels).
      \item Associated production with other particles.
    \end{itemize}
    \item Plays large role in electroweak symmetry breaking and sensitive to BSM physics.
    \item LHC is a top factory.
  \end{itemize}
\end{frame}


% ========== TOP PAIR PRODUCTION ==========
\begin{frame}{$t\bar{t}$ Production}
  \begin{itemize}
    \item Dominant production mechanism at LHC.
    \item Mediated by strong interaction.
    \item Produced by either gluon-gluon fusion (90\%) or quark-antiquark annihilation (10\%).
  \end{itemize}
\end{frame}


% ========== SINGLE TOP PRODUCTION ==========

  \begin{itemize}
    \item Charged-current electroweak process.
    \item Three production channels:
    \begin{itemize}
      \item $t$-channel: $q b \rightarrow q' t$
      \item $s$-channel: $q \bar{q}' \rightarrow t \bar{b}$
      \item $W$-associated production ($tW$): $g b \rightarrow t W$
    \end{itemize}
  \end{itemize}

% ========== ASSOCIATED TOP PRODUCTION ==========
\begin{frame}{Associated Top Production}
  \begin{itemize}
    \item Top quarks can also be produced in association with other particles, such as:
    \begin{itemize}
      \item $t\bar{t}W$ (top quark pair with W boson)
      \item $t\bar{t}Z$ (top quark pair with Z boson)
      \item $t\bar{t}\gamma$ (top quark pair with photon)
      \item $t\bar{t}H$ (top quark pair with Higgs boson)
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{$t\bar{t}W$ Cross Section}
  \begin{itemize}
    \item ATLAS: Inclusive cross section $\sigma_{t\bar{t}W} = 880 \pm 80$fb, in agreement with SM prediction within 1.4$\sigma$.
    \begin{itemize}
      \item Run 2 of LHC, $\sqrt{s} = \SI{13}{\tera\electronvolt}$, $L_{int} = \SI{140}{\per\femto\barn}$.
      \item Relative charge asymmetry also agrees with SM prediction.
    \end{itemize}
    \item CMS: Inclusive cross section $\sigma_{t\bar{t}W} = 868 \pm 40 (\text{stat}) \pm 51 (\text{syst})$ fb, within 2$\sigma$ of SM prediction. 
    \begin{itemize}
      \item Run 2 of LHC, $\sqrt{s} = \SI{13}{\tera\electronvolt}$, $L_{int} = \SI{138}{\per\femto\barn}$.
      \item Ratio of $\sigma_{t\bar{t}W^+}$ to $\sigma_{t\bar{t}W^-}$ measured to be $1.61 \pm 0.15\,\text{(stat)}^{+0.07}_{-0.05}\,\text{(syst)}$, consistent with SM within 2$\sigma$.
    \end{itemize}
  \end{itemize}
    {\tiny \color{gray} ATLAS Collaboration, \textbf{\href{https://doi.org/10.1007/JHEP07(2023)219}{JHEP 07 (2023) 219}} [\href{https://arxiv.org/abs/2208.06485v2}{\text{arXiv:2208.06485v2}}]}
\end{frame}

\begin{frame}{$t\bar{t}+Z$ Cross Section}
  \begin{itemize}
    \item Inclusive cross section measured at $\sigma_{t\bar{t}Z} = 0.86 \pm 0.04 (\text{stat}) \pm 0.04 (\text{syst})$ pb.
    \begin{itemize}
      \item LHC Run 2, $\sqrt{s} = \SI{13}{\tera\electronvolt}$, $L_\text{int} = \SI{140}{\per\femto\barn}$.
    \end{itemize}
    \item In agreement with Standard Model NLO + NNLL prediction.
    \item Two, three and four lepton final states considered.
  \end{itemize}
