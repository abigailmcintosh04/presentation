\section{Top Physics}
\subsection{Introduction}

% ========== TOP INTRODUCTION ==========
\begin{frame}{Top Quark}
  \begin{itemize}
    \item Heaviest known elementary particle -- $m_t = 172.6 \pm 0.3\,\unit{\giga\electronvolt}$.
    \item Yukawa coupling $y_t \sim 1$, largest of all fermions.
    \item Short lifetime ($\tau \sim \qty{5e-25}{\second}$), decays before hadronisation.
    \item Plays large role in electroweak symmetry breaking and sensitive to BSM physics.
    \item LHC is a top factory.
    \item Main production mechanisms:
    \begin{itemize}
      \item Pair production via strong interaction.
      \item Single top production via electroweak interaction.
      \item Associated $t\bar{t}$ production.
    \end{itemize}
  \end{itemize}
\end{frame}

    \begin{itemize}
      \item $t$-channel: $q b \rightarrow q' t$
      \item $s$-channel: $q \bar{q}' \rightarrow t \bar{b}$
      \item $W$-associated production ($tW$): $g b \rightarrow t W$

\begin{frame}{Single Top Production Cross Sections}
  \begin{itemize}
    \item $t$-channel (ATLAS): \textcolor{blue}{$\sigma_{\text{$t$-ch}} = 221 \pm 13\,\unit{\pico\barn}$}.
    \begin{itemize}
      \item Ratio of cross sections for single top and antitop \textcolor{blue}{$R_{\text{$t$-ch}} = 1.636 {}^{+0.036}_{-0.034}$}.
    \end{itemize}
    \item $s$-channel (ATLAS): \textcolor{blue}{$\sigma_{\text{$s$-ch}} = 10.8 \pm 1.5\,\text{(stat.)} \pm 1.6\,\text{(syst.)}\,\unit{\pico\barn}$}.

    \item All cross-section measurements taken from ATLAS or CMS at LHC Run 2, $\sqrt{s} = \qty{13}{\tera\electronvolt}$, $L_{int} \approx \qty{140}{\per\femto\barn}$.
  \end{itemize}

\subsubsection{Associated $t\bar{t}$ Production}

% ========== ASSOCIATED TOP PRODUCTION ==========
\begin{frame}{Associated $t\bar{t}$ Production}
  \begin{itemize}
    \item Top quarks can be produced as a pair in association with other particles, such as:
    \begin{itemize}
      \item $t\bar{t}W$ (top quark pair with W boson)
      \item $t\bar{t}Z$ (top quark pair with Z boson)
      \item $t\bar{t}\gamma$ (top quark pair with photon)
      \item $t\bar{t}H$ (top quark pair with Higgs boson)
      \item Other various processes, e.g. $t\bar{t}t\bar{t}$ (four top quarks) and diboson production (not covered).
    \end{itemize}
    \item Deviations from SM predictions could indicate new effects in electroweak symmetry breaking.
    \item All cross-section measurements taken from ATLAS or CMS at LHC Run 2, $\sqrt{s} = \qty{13}{\tera\electronvolt}$, $L_{int} \approx \qty{140}{\per\femto\barn}$.
  \end{itemize}
\end{frame}

\section{Effective Field Theory}
\subsection{Introduction}

% ==================== EFT INTERPRETATION ====================
\begin{frame}{Effective Field Theory}
  \begin{itemize}
    \item Effective field theories (EFT) provide a model-independent framework allowing for for the study of BSM phemomena at currently unreachable energy scales (denoted $\Lambda$).
    \item Allows properties of high-energy BSM physics at scale $\Lambda$ to be related with low-energy measurements.
    \item The dimensionless Wilson coefficients $c_i^{(d)}$ are used to parameterise the strength of the new physics effects.
  \end{itemize}  
  \begin{block}{SMEFT Lagrangian}
    \begin{equation}
      \mathcal{L} = \mathcal{L}_\text{SM} + \sum_{d=5}^{\infty} \sum_i \frac{c_i^{(d)}}{\Lambda^{d-4}} \mathcal{O}_i^{(d)},
    \end{equation}
    where $d$ is mass dimension, $c_i^{(d)}$ are the Wilson coefficients, $\Lambda$ is the energy scale of new physics, and $\mathcal{O}_i^{(d)}$ are the higher-dimensional operators.
  \end{block}
\end{frame}

\subsection{SMEFT}

\begin{frame}{SMEFT}
  \begin{itemize}
    \item SMEFT extends the SM Lagrangian with higher-dimensional operators, suppressed by powers of $\Lambda$.
    \item Typically interested in dimension-6 operators.
    \begin{itemize}
      \item SM contains dimension-2 and dimension-4 operators.
      \item Dimension-5 operators provide a mechanism for neutrino mixing.
    \end{itemize}
    \item The Warsaw basis is a commonly used non-redundant set of dimension-6 operators.
    \begin{itemize}
      \item Many of them, but only a few are relevant for top quark physics.
    \end{itemize}
    \item Preserves the SM gauge group $SU(3)_C \times SU(2)_L \times U(1)_Y$.
  \end{itemize}
  {\tiny \color{gray} University of Warsaw, \textbf{\href{https://doi.org/10.1007/JHEP10(2010)085}{JHEP 10 (2010) 085}} [\href{https://arxiv.org/abs/1008.4884v3}{\texttt{arXiv:1008.4884}}]}
\end{frame}

\subsubsection{EFT Effects}

\begin{frame}{EFT Effects}
  \begin{itemize}
    \item SMEFT operators can modify SM predictions of rates and kinematic distributions of certain processes.
    \item SMEFT operators can also allow SM-forbidden processes to occur.
    \item Precise top measurements allows for constraints to be placed on the relevant Wilson coefficients of the SMEFT operators. 
  \end{itemize}
  {\tiny \color{gray} J. Wilson for ATLAS \& CMS, TOP2022 \textbf{\href{https://conference.ippp.dur.ac.uk/event/925/contributions/5716/}{(Indico)}}}
\end{frame}

\subsubsection{EFT Interpretations}

\begin{frame}{EFT Interpretations of top + X}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \begin{block}{Detector-level fit}
        \begin{itemize}
          \item Paramaterise predictions of EFT contributions directly at detector level.
          \item Simplest approach as long as bins are statistically independent, and all detector-level effects are accounted for.
          \item Inflexible in reinterpetation of results, as any update in theory requires full rerun of detector simulation.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Unfolding}
        \begin{itemize}
          \item Estimate EFT contributions by undoing detector effects to return to particle/parton level.
          \item Simple to reinterpret as full detector simulation not required.
          \item Some EFT coefficients can impact signal and background processes, meaning traditional separation of the two may require modification. 
          \vspace{\baselineskip}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  {\tiny \color{gray} ATLAS Collaboration, \href{https://cds.cern.ch/record/2872789/}{\texttt{ATL-PHYS-PUB-2023-030}}}
\end{frame}

\subsection{Future Outlook}

\begin{frame}{Future Outlook}
  \begin{itemize}
    \item LHC Run 3 increased integrated luminosity ($\sim \qty{355}{\per\femto\barn}$) allows for more precise top quark measurements and makes it easier to probe rare processes.
    \item Machine learning techniques are being increasingly used to analyse LHC data.
    \begin{itemize}
      \item Identify and separate signal processes from SM backgrounds.
      \item Suppress migrations in unfolding.
      \item Stronger constraints on dimension-six operators through improved scaling of likelihood ratio estimations.
      \item Integration of unbinned multivariate observables into SMEFT fits, preventing loss of information due to binning.
    \end{itemize}
  \end{itemize}
  {\tiny \color{gray} NIKHEF, \textbf{\href{https://doi.org/10.1007/JHEP03(2023)033}{JHEP 03 (2023) 033}} [\href{https://arxiv.org/abs/2211.02058v2}{\texttt{arXiv:2211.02058}}]} \\
  {\tiny \color{gray} J. Brehmer et al., \textbf{\href{https://doi.org/10.1103/PhysRevLett.121.111801}{Phys. Rev. Lett. 121 (2018) 111801}} [\href{https://arxiv.org/abs/1805.00013v2}{\texttt{arXiv:1805.00013}}]} \\
  {\tiny \color{gray} ATLAS Collaboration, \href{https://cds.cern.ch/record/2872789/}{\texttt{ATL-PHYS-PUB-2023-030}}}
\end{frame}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item Top quark measurements sensitive to BSM physics.
    \item Deviations from SM predictions in associated top production channels could indicate new EWSB effects.
    \item EFT allows for study of BSM phenomena at currently unreachable energy scales in a model-independent way.
    \item Precise top measurements allow for constraints on Wilson coefficients of SMEFT operators.
    \item LHC Run 3 and machine learning techniques should allow for more precise top measurements and stronger constraints on SMEFT operators.
  \end{itemize}
\end{frame}
